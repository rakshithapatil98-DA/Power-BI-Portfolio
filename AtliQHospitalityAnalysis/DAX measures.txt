DAX Measures -->>

1) Revenue = SUM(fact_bookings[revenue_realized])
2) Total Bookings = COUNT(fact_bookings[booking_id])
3) Total Cancelled Bookings = CALCULATE(COUNT(fact_bookings[booking_id]),fact_bookings[booking_status]="cancelled")
4) Total Capacity = SUM(fact_aggregated_bookings[capacity])
5) Total checked out = CALCULATE(SUM(fact_bookings[booking_id]),fact_bookings[booking_status]="Checked out")
6) Total No Shows = CALCULATE(COUNT(fact_bookings[booking_id]),fact_bookings[booking_status]="No Show")
7) Total Successful bookings = SUM(fact_aggregated_bookings[successful_bookings])
8) No of Days = COUNT(dim_date[date])
9) No Show % = DIVIDE([Total No Shows],[Total Bookings],0)
10) Occupancy % = SUM(fact_aggregated_bookings[successful_bookings])/SUM(fact_aggregated_bookings[capacity])
11) Realisation % = 1-([Cancellation Rate%]+[No Show %])
12) Revpar = DIVIDE([Revenue],[Total Capacity],0)
13) URN = DIVIDE([Total checked out],[No of Days],0)
14) BRN = DIVIDE([Total Bookings],[No of Days],0)
15) DSRN = DIVIDE([Total Capacity],[No of Days],0)
16) Chng % = ([Revenue]-[LY Revenue])/[LY Revenue]
17) Cancellation Rate% = DIVIDE([Total Cancelled Bookings],[Total Bookings],0)
18) Bookings% by Platform = DIVIDE([Total Bookings],CALCULATE([Total Bookings],ALL(fact_bookings[booking_platform])))
19) Avg Rating = AVERAGE(fact_bookings[ratings_given])
20) ADR = DIVIDE([Revenue],[Total Bookings],0)
